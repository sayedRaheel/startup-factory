package main

import (
	"fmt"
	"os"
)

func main() {
	if len(os.Args) < 2 {
		fmt.Println("Usage: vise <prompt> | vise init")
		os.Exit(1)
	}

	if os.Args[1] == "init" {
		if err := initAivise(); err != nil {
			fmt.Printf("Error: %v\n", err)
			os.Exit(1)
		}
		fmt.Println("✔ Generated .aivise configuration.")
		return
	}

	prompt := os.Args[1]

	config, err := loadAivise()
	if err != nil {
		fmt.Printf("Error loading config: %v\n", err)
		os.Exit(1)
	}

	ctxPayload, err := compileWorkspaceContext()
	if err != nil {
		fmt.Printf("Error compiling context: %v\n", err)
		os.Exit(1)
	}

	fmt.Println("Compiling context and generating diff...")
	aiResp, err := executePrompt(prompt, ctxPayload, config)
	if err != nil {
		fmt.Printf("Error executing prompt: %v\n", err)
		os.Exit(1)
	}

	fmt.Println("Applying diff to memory and linting...")
	backup, err := applyToMemory(aiResp)
	if err != nil {
		fmt.Printf("Error applying to memory: %v\n", err)
		os.Exit(1)
	}

	success, err := runToolchain(config.LintCommand)
	if err != nil {
		fmt.Printf("Toolchain execution error: %v\n", err)
	}

	if success {
		fmt.Println("✔ Lint passed. Changes committed.")
	} else {
		fmt.Println("✖ Lint failed. AI hallucinated. Rolling back changes.")
		rollback(backup)
		os.Exit(1)
	}
}

package main

import (
	"os"
	"github.com/pelletier/go-toml/v2"
)

type AiviseConfig struct {
	Language    string   `toml:"language"`
	LintCommand string   `toml:"lint_command"`
	StrictRules []string `toml:"strict_rules"`
}

func initAivise() error {
	cfg := AiviseConfig{
		Language:    "go",
		LintCommand: "go build ./...",
		StrictRules: []string{
			"Never use panic().",
			"Always add inline documentation for public functions.",
		},
	}
	b, err := toml.Marshal(cfg)
	if err != nil {
		return err
	}
	return os.WriteFile(".aivise", b, 0644)
}

func loadAivise() (AiviseConfig, error) {
	b, err := os.ReadFile(".aivise")
	if err != nil {
		// Fallback to defaults if missing
		return AiviseConfig{
			Language:    "unknown",
			LintCommand: "echo 'No lint'",
			StrictRules: []string{},
		}, nil
	}
	var cfg AiviseConfig
	err = toml.Unmarshal(b, &cfg)
	return cfg, err
}

package main

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"

	ignore "github.com/sabhiram/go-gitignore"
)

func compileWorkspaceContext() (string, error) {
	var contextBuilder strings.Builder
	ignorer, _ := ignore.CompileIgnoreFile(".gitignore")

	err := filepath.WalkDir(".", func(path string, d os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		
		// Skip standard noisy directories explicitly
		if d.IsDir() && (d.Name() == ".git" || d.Name() == "node_modules" || d.Name() == ".venv") {
			return filepath.SkipDir
		}

		if ignorer != nil && ignorer.MatchesPath(path) {
			if d.IsDir() {
				return filepath.SkipDir
			}
			return nil
		}

		if !d.IsDir() {
			content, err := os.ReadFile(path)
			if err == nil {
				// Naive binary check: avoid appending large non-text files
				if !strings.Contains(httpDetectContentType(content), "text") && !strings.HasSuffix(path, ".go") {
				    return nil
				}
				contextBuilder.WriteString(fmt.Sprintf("--- FILE: %s ---\n%s\n", path, string(content)))
			}
		}
		return nil
	})

	return contextBuilder.String(), err
}

// Simple fallback heuristic for context.go
func httpDetectContentType(b []byte) string {
    if len(b) == 0 {
        return "text/plain"
    }
    return "text" // Simplified for prototype context compilation
}
EOF

package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
)

type FileChange struct {
	Path    string `json:"path"`
	Content string `json:"content"`
}

type AiResponse struct {
	Files []FileChange `json:"files"`
}

func executePrompt(prompt, context string, config AiviseConfig) (*AiResponse, error) {
	apiKey := os.Getenv("OPENAI_API_KEY")
	if apiKey == "" {
		fmt.Println("Warning: OPENAI_API_KEY not set. Returning a deterministic mock response.")
		// Mock deterministic behaviour to ensure system can be tested offline
		return &AiResponse{
			Files: []FileChange{},
		}, nil
	}

	systemPrompt := fmt.Sprintf(
		"You are an infallible code compiler. Rules: %v. Context: %s. OUTPUT STRICTLY JSON. NO MARKDOWN. NO CHAT. Format: { \"files\": [ { \"path\": \"str\", \"content\": \"str\" } ] }",
		config.StrictRules, context,
	)

	reqBody := map[string]interface{}{
		"model": "gpt-4o",
		"response_format": map[string]string{
			"type": "json_object",
		},
		"messages": []map[string]string{
			{"role": "system", "content": systemPrompt},
			{"role": "user", "content": prompt},
		},
	}

	b, err := json.Marshal(reqBody)
	if err != nil {
		return nil, err
	}

	req, err := http.NewRequest("POST", "https://api.openai.com/v1/chat/completions", bytes.NewReader(b))
	if err != nil {
		return nil, err
	}
	req.Header.Set("Authorization", "Bearer "+apiKey)
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, err
	}

	var oaiResp struct {
		Choices []struct {
			Message struct {
				Content string `json:"content"`
			} `json:"message"`
		} `json:"choices"`
	}

	if err := json.Unmarshal(body, &oaiResp); err != nil {
		return nil, fmt.Errorf("failed to parse API wrapper: %w\nBody: %s", err, string(body))
	}

	if len(oaiResp.Choices) == 0 {
		return nil, fmt.Errorf("no choices returned from API")
	}

	var aiResp AiResponse
	if err := json.Unmarshal([]byte(oaiResp.Choices[0].Message.Content), &aiResp); err != nil {
		return nil, fmt.Errorf("failed to parse strict JSON from AI: %w", err)
	}

	return &aiResp, nil
}

package main

import (
	"os"
	"os/exec"
	"strings"
)

func runToolchain(commandStr string) (bool, error) {
	parts := strings.Fields(commandStr)
	if len(parts) == 0 {
		return true, nil
	}

	cmd := exec.Command(parts[0], parts[1:]...)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr

	err := cmd.Run()
	return err == nil, nil
}

cat << 'EOF' > workspace.go
package main

import (
	"os"
	"path/filepath"
)

const DeletedSentinel = "\x00DELETED\x00"

type Backup struct {
	OriginalFiles []FileChange
}

func applyToMemory(response *AiResponse) (*Backup, error) {
	backup := &Backup{}

	for _, file := range response.Files {
		// Backup existing
		content, err := os.ReadFile(file.Path)
		if err == nil {
			backup.OriginalFiles = append(backup.OriginalFiles, FileChange{
				Path:    file.Path,
				Content: string(content),
			})
		} else {
			// File didn't exist
			backup.OriginalFiles = append(backup.OriginalFiles, FileChange{
				Path:    file.Path,
				Content: DeletedSentinel,
			})
		}

		// Write new file
		dir := filepath.Dir(file.Path)
		if err := os.MkdirAll(dir, 0755); err != nil {
			return backup, err
		}
		if err := os.WriteFile(file.Path, []byte(file.Content), 0644); err != nil {
			return backup, err
		}
	}

	return backup, nil
}

func rollback(backup *Backup) error {
	for _, file := range backup.OriginalFiles {
		if file.Content == DeletedSentinel {
			os.Remove(file.Path)
		} else {
			os.WriteFile(file.Path, []byte(file.Content), 0644)
		}
	}
	return nil
}
EOF
